% S3 Frame 5: The body — branching packs area into volume
% ---------------------------------------------------------------
\begin{frame}{Your body branches to pack a huge surface into a small space}
  \begin{columns}[c]
    \begin{column}{0.58\textwidth}
      \centering
      \includegraphics[width=0.62\textwidth]{images/part3/retina.png}\hfill
      \raisebox{0.14\textwidth}{\includegraphics[width=0.34\textwidth]{images/part3/retina_zoom.png}}\\[2pt]
      {\footnotesize\color{DarkGray} blood vessels in a human eye, and a zoom-in}
    \end{column}
    \begin{column}{0.42\textwidth}
      \small
      \begin{itemize}
        \item Lungs: about \highlight{23 splits}\\
          $\Rightarrow$ $\sim$\highlight{70 m$^2$} of surface\\
          inside $\sim$5 litres
        \item Vessels: every cell within reach, using little space
        \item Doctors track the \highlight{dimension} of eye vessels to spot disease early
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.05cm}
  \takeaway{Branching again and again is nature's cheapest way to get \highlight{area out of volume}.}

  \source{Photo: M.~Häggström, ``Fundus photograph of normal left eye'', 2014, CC0. Lung figures: Weibel (1963).}
\end{frame}

% ---------------------------------------------------------------
% S3 Frame 6: Databases — space-filling curve = spatial index
% ---------------------------------------------------------------
\begin{frame}{A curve that fills a square turns a map into a sorted list}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=0.95\textwidth]{images/part3/hilbert_index.png}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \begin{itemize}
        \item A database can only sort by \highlight{one number}
        \item Walk the \highlight{Hilbert curve} through the map: each cell gets its position on the curve
        \item Nearby places $\rightarrow$ nearby numbers $\rightarrow$ one quick range search
      \end{itemize}
      \vspace{0.2cm}
      \takeaway{This trick powers map search in systems like Google~S2 and Uber~H3.}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------
% S3 Frame 7: Game worlds — grown from a rule, not stored
% ---------------------------------------------------------------
\begin{frame}{Game worlds are grown from a rule, not stored}
  \centering
  \includegraphics[width=0.98\textwidth]{images/part3/terrain_octaves.png}

  \vspace{0.15cm}
  \takeaway{Add \highlight{smaller bumps at every scale} --- the Koch idea --- and a
    handful of numbers becomes an endless mountain range. Store the rule, not the map.}

  \source{Terrain generated here by summing 7 layers of noise, each half the size of the last (fractal Brownian motion).}
\end{frame}

% ---------------------------------------------------------------
% S3 Frame 8: Limits — real fractals stop
% ---------------------------------------------------------------
\begin{frame}{Real fractals stop at some scale}
  \begin{columns}[c]
    \begin{column}{0.66\textwidth}
      \centering
      \includegraphics[width=\textwidth]{images/part3/richardson_limit.png}
    \end{column}
    \begin{column}{0.33\textwidth}
      \small
      \begin{itemize}
        \item Coast: bays look like bays, but \highlight{sand grains don't}
        \item Lungs: stop after $\sim$23 splits
        \item Only the Sierpinski triangle goes on \highlight{forever}
      \end{itemize}
      \vspace{0.2cm}
      \takeaway{A fractal is a \highlight{model} of nature --- and every model has a range where it works.}
    \end{column}
  \end{columns}
\end{frame}
