\begin{frame}{The Self-Similarity Rule}
  \centering
  \small
  After scaling by factor $r$, the number of self-similar pieces is:
  \quad
  \highlight{\Large $N = \left(\dfrac{1}{r}\right)^{\!D}$}

  \vspace{0.25cm}
  \only<1>{%
    {\normalsize \textbf{\color{PrimaryColor}Square ($D = 2$)} \quad with scale factor $r = \frac{1}{2}$:\quad
    $N = \left(\frac{1}{1/2}\right)^{\!2} = 2^2 = \mathbf{4\text{ pieces}}$}%
  }%
  \only<2>{%
    {\normalsize \textbf{\color{PrimaryColor}Cube ($D = 3$)} \quad with scale factor $r = \frac{1}{2}$:\quad
    $N = \left(\frac{1}{1/2}\right)^{\!3} = 2^3 = \mathbf{8\text{ pieces}}$}%

        \begin{conceptbox}{Step 1}
          Scale the Sierpinski Triangle by $\frac{1}{2}$.
        \end{conceptbox}
      }
      \only<2>{
        \begin{conceptbox}{Step 2 --- Self-Similar Pieces}
          We get exactly \highlight{3} self-similar copies
          of the original triangle.

          \vspace{0.2cm}
          Each smaller triangle is a perfect replica,
          scaled by $r = \frac{1}{2}$.
        \end{conceptbox}
      }
      \only<3>{
        \begin{mathbox}{Step 3 --- The Equation}
          Using $N = \left(\frac{1}{r}\right)^D$:
          \begin{align*}
            3 &= \left(\frac{1}{1/2}\right)^{\!D} \\[6pt]
            3 &= 2^D
          \end{align*}
        \end{mathbox}
      }
      \only<4>{
        \begin{mathbox}{Step 4 --- Solving for $D$}
          \begin{align*}
            2^D &= 3 \\[4pt]
            D\,\log 2 &= \log 3 \\[4pt]
            D &= \frac{\log 3}{\log 2}
          \end{align*}
          \begin{center}
            {\fontsize{22}{28}\selectfont
              $D \approx \highlight{1.585}$}
          \end{center}
        \end{mathbox}

        \vspace{0.1cm}
        \highlight{Not an integer!} --- strictly between
        1D (line) and 2D (surface).
      }

% S2 Frame 8: The General Theory — Box-Counting Dimension
% ---------------------------------------------------------------
\begin{frame}{The General Theory: Box-Counting Dimension}
  \small
  The \highlight{Self-Similarity Dimension} works beautifully for
  self-similar fractals. But what about more complex shapes?

  \vspace{0.3cm}
  \begin{mathbox}{Box-Counting Dimension}
    The fractal dimension $D$ is defined as:
    $$D = \lim_{\varepsilon \to 0}
      \frac{\log N(\varepsilon)}{\log(1/\varepsilon)}$$
    where $N(\varepsilon)$ is the number of boxes of side length
    $\varepsilon$ needed to cover the fractal.
  \end{mathbox}

  \vspace{0.3cm}
  \pause
  \begin{conceptbox}{Intuition}
    \begin{itemize}
      \item Cover the fractal with a grid of boxes of size
        $\varepsilon$
      \item Count how many boxes $N(\varepsilon)$ the fractal
        touches
      \item Shrink $\varepsilon \to 0$ and observe how
        $N(\varepsilon)$ grows
      \item The \highlight{rate of growth} tells us the
        dimension
    \end{itemize}
  \end{conceptbox}

  \vspace{0.2cm}
  \begin{center}
    {\footnotesize\color{DarkGray} For a deeper dive, see
    \href{https://www.youtube.com/watch?v=gB9n2gHsHN4}%
    {\textcolor{AccentColor}{3Blue1Brown:
    Fractals are typically not self-similar}}}
  \end{center}
\end{frame}
